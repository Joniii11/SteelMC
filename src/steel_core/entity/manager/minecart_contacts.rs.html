16// Both concrete implementations delegate their push callback to AbstractMinecart.
17fn uses_native_push(entity: &dyn Entity) -> bool {
18    entity.is::<MinecartEntity>() || entity.is::<TntMinecartEntity>()
19}

21pub(super) struct CoincidentCartQuery {
22    bounds: EntitySpatialCellBounds,
23    /// None caches a mixed or empty region until a relevant input changes.
24    common_xz: Option<[u64; 2]>,
25    valid: Arc<AtomicBool>,
26}
27
28impl Drop for CoincidentCartQuery {
29    fn drop(&mut self) {
30        // Readers may still hold a token while checking the world's multipart index.
31        self.valid.store(false, Ordering::Release);
32    }
33}
34
35impl CoincidentCartQuery {
36    fn inspect(state: &ManagerState, hitbox: &WorldAabb, bounds: EntitySpatialCellBounds) -> Self {
37        let valid = Arc::new(AtomicBool::new(true));
38        // Include every entry in the broad-phase cells regardless of accessibility or
39        // exact bounds. Motion within those cells cannot introduce an unobserved candidate.
40        let entries =
41            WorldEntityManager::query_entries(state, hitbox, EntityCollisionCandidates::All, None);
42        let mut positions = entries.map(|entry| {
43            let entity = entry.entity.as_ref();
44            if !uses_native_push(entity) {
45                return None;
46            }
47            entity.base().observe_minecart_contacts(&valid)
48        });
49        let common_xz = positions
50            .next()
51            .flatten()
52            .filter(|xz| positions.all(|other| other == Some(*xz)));
53        Self {
54            bounds,
55            common_xz,
56            valid,
57        }
58    }
59
60    fn for_source(&self, state: &ManagerState, source: &dyn Entity) -> Option<Arc<AtomicBool>> {
61        let xz = self.common_xz?;
62        let live = state.live_by_id.get(source.id())?.entity.as_ref();
63        if !ptr::addr_eq(live, source)
64            || source.base().observe_minecart_contacts(&self.valid) != Some(xz)
65        {
66            return None;
67        }
68        self.valid
69            .load(Ordering::Acquire)
70            .then(|| Arc::clone(&self.valid))
71    }
72}
73
74impl ManagerState {
75    fn current_cart_query(&self, bounds: EntitySpatialCellBounds) -> Option<&CoincidentCartQuery> {
76        self.coincident_cart_query
77            .as_ref()
78            .filter(|query| query.bounds == bounds && query.valid.load(Ordering::Acquire))
79    }
80}
81
82impl WorldEntityManager {
83    /// On success all indexed candidates share the source's exact X/Z and use the
84    /// native minecart push. Recheck the returned token after consulting any other index.
85    pub(crate) fn coincident_minecarts(
86        &self,
87        hitbox: &WorldAabb,
88        source: &dyn Entity,
89    ) -> Option<Arc<AtomicBool>> {
90        if !uses_native_push(source) {
91            return None;
92        }
93        let bounds = EntitySpatialCellBounds::from_aabb(hitbox);
94        if bounds.direct_probe_count() > MAX_DIRECT_SPATIAL_CELL_PROBES {
95            return None;
96        }
97        {
98            let state = self.state.read();
99            if let Some(query) = state.current_cart_query(bounds) {
100                return query.for_source(&state, source);
101            }
102            // An isolated cart doesn't benefit from a cache allocation or a manager write lock.
103            if !bounds.cells().iter().any(|cell| {
104                state
105                    .by_spatial_cell
106                    .get(cell)
107                    .is_some_and(|entries| entries.entities.len() > 1)
108            }) {
109                return None;
110            }
111        }
112        let mut state = self.state.write();
113        if state.current_cart_query(bounds).is_none() {
114            state.coincident_cart_query =
115                Some(CoincidentCartQuery::inspect(&state, hitbox, bounds));
116        }
117        state
118            .coincident_cart_query
119            .as_ref()?
120            .for_source(&state, source)
121    }
